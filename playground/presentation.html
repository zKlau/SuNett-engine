import { SunettEngine } from
  "@zklau/sunett-engine";
import type { Song } from
  "@zklau/sunett-engine";

export async function renderTab(song: Song) {
  const engine = new SunettEngine();

  await engine.loadSong(song);
  engine.render(0, { target: "#tabs" });
  engine.enableSelectionInput();

  return engine;
}
